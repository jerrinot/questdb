cmake_minimum_required(VERSION 3.22)
project(questdb_query_codegen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
# Use the exact dependency selected by QuestDB, independently of stale local build trees.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../core/CMakeLists.txt" QUESTDB_CMAKE)
string(REGEX MATCH "FetchContent_Declare\\([ \r\n]+asmjit[^)]*\\)" ASMJIT_DECLARATION "${QUESTDB_CMAKE}")
string(REGEX MATCH "GIT_REPOSITORY[ \r\n]+([^ \r\n]+)" UNUSED "${ASMJIT_DECLARATION}")
set(ASMJIT_REPOSITORY "${CMAKE_MATCH_1}")
string(REGEX MATCH "GIT_TAG[ \r\n]+([0-9a-f]+)" UNUSED "${ASMJIT_DECLARATION}")
set(ASMJIT_REVISION "${CMAKE_MATCH_1}")
if(NOT ASMJIT_REVISION OR NOT ASMJIT_REPOSITORY)
    message(FATAL_ERROR "Could not read QuestDB's AsmJit dependency pin")
endif()
include(FetchContent)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(ASMJIT_EMBED OFF CACHE BOOL "" FORCE)
set(ASMJIT_NO_FOREIGN ON CACHE BOOL "" FORCE)
FetchContent_Declare(asmjit GIT_REPOSITORY "${ASMJIT_REPOSITORY}" GIT_TAG "${ASMJIT_REVISION}")
FetchContent_MakeAvailable(asmjit)
add_library(query_codegen SHARED asmjit_backend.cpp)
target_link_libraries(query_codegen PRIVATE asmjit::asmjit)
target_compile_options(query_codegen PRIVATE -Wall -Wextra -Werror)

# LLVM is required only by this opt-in experimental module, never by the normal build.
find_package(LLVM 21.1 REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 21)
    message(FATAL_ERROR "The experimental LLVM bridge is tested with LLVM 21")
endif()
add_library(query_codegen_llvm SHARED llvm_backend.cpp)
target_include_directories(query_codegen_llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITION_FLAGS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(query_codegen_llvm PRIVATE ${LLVM_DEFINITION_FLAGS})
target_link_libraries(query_codegen_llvm PRIVATE LLVM)
target_compile_options(query_codegen_llvm PRIVATE -Wall -Wextra -Werror)
